function renderTasks(){
  const list = document.getElementById('taskList');
  const doneList = document.getElementById('doneList');
  list.innerHTML = ''; doneList.innerHTML = '';
  tasks.forEach(t=>{
    const el = document.createElement('div');
    el.className = 'glass task-item' + (t.done?' done':'');
    let html = '<div class="checkbox"></div><div style="flex:1;">';
    html += '<div class="task-text">'+t.text+'</div>';
    if(t.breakdown && !t.done){
      html += '<button class="breakdown-btn">AI 拆成小步骤</button>';
      if(t._showBreak){
        html += '<div class="breakdown-list">';
        t.breakdown.forEach(s=> html += '<div class="sub-task">'+s+'</div>');
        html += '</div>';
      }
    }
    html += '</div>';
    el.innerHTML = html;
    el.onclick = (e)=>{
      if(e.target.closest('.breakdown-btn')) return;
      const wasDone = t.done;
      t.done=!t.done; t._showBreak=false; save(); renderTasks(); renderProgress();
      if(!wasDone && t.done){
        ipSpeak(IP_DONE[Math.floor(Math.random()*IP_DONE.length)], 'assets/ip_happy.png');
        ipBounce();
        setTimeout(()=>{ ipSpeak('先做五分钟，就从这一步开始。', 'assets/ip_normal.png'); }, 2500);
      }
    };
    const bb = el.querySelector('.breakdown-btn');
    if(bb) bb.onclick = (e)=>{ e.stopPropagation(); t._showBreak=!t._showBreak; renderTasks(); toast(t._showBreak?'已拆成 5 个 5 分钟小步骤':'已收起'); };
    (t.done?doneList:list).appendChild(el);
  });
}

let focusSec = 300, focusTimer = null;
const CIRC = 660;
function renderFocus(){
  const m = Math.floor(focusSec/60), s = focusSec%60;
  document.getElementById('focusTime').textContent = String(m).padStart(2,'0')+':'+String(s).padStart(2,'0');
  document.getElementById('focusRing').style.strokeDashoffset = CIRC*(1-focusSec/300);
}
document.getElementById('focusBtn').onclick = ()=>{
  const btn = document.getElementById('focusBtn');
  if(focusTimer){
    clearInterval(focusTimer); focusTimer=null;
    btn.textContent='继续 5 分钟'; document.getElementById('focusState').textContent='已暂停';
  } else {
    focusTimer = setInterval(()=>{
      focusSec--;
      if(focusSec<=0){
        clearInterval(focusTimer); focusTimer=null; focusSec=300;
        btn.textContent='再专注一次'; document.getElementById('focusState').textContent='完成了，做得好';
        toast('5 分钟到了，休息一下吧');
      }
      renderFocus();
    },1000);
    btn.textContent='暂停'; document.getElementById('focusState').textContent='专注中…启启在陪着你';
  }
};
document.getElementById('focusExit').onclick = ()=>{
  if(focusTimer){ clearInterval(focusTimer); focusTimer=null; }
  focusSec=300; renderFocus();
  document.getElementById('focusBtn').textContent='开始 5 分钟';
  document.getElementById('focusState').textContent='准备开始';
  toast('已退出专注，随时可以再回来');
};
renderFocus();

}
